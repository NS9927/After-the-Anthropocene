on_actions = {

    on_startup = {
        effect = {
            every_country = {
                limit = { is_ai = no }
                ata_refresh_party_list = yes
            }
        }
    }

    on_daily = {
        effect = {
            if = {
                limit = { is_ai = no }
                ata_refresh_party_list = yes
            }
        }
    }
}
